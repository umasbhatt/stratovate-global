    <script>

        const teamInput = document.getElementById('team-size');

        const hoursInput = document.getElementById('hours');

        const effInput = document.getElementById('efficiency');


        const teamVal = document.getElementById('team-size-val');

        const hoursVal = document.getElementById('hours-val');

        const effVal = document.getElementById('efficiency-val');


        const resultHours = document.getElementById('result-hours');

        const resultValue = document.getElementById('result-value');


        function updateCalculator() {

            const team = parseInt(teamInput.value);

            const hours = parseInt(hoursInput.value);

            const eff = parseInt(effInput.value) / 100;


            teamVal.textContent = team.toLocaleString();

            hoursVal.textContent = hours;

            effVal.textContent = Math.round(eff * 100);


            const savedHoursPerWeek = team * hours * eff;

            const savedHoursPerYear = Math.round(savedHoursPerWeek * 52);

            const valueEquiv = Math.round(savedHoursPerYear * 30);


            resultHours.textContent = savedHoursPerYear.toLocaleString() + ' hrs';

            resultValue.textContent = '$' + valueEquiv.toLocaleString() + ' / yr';

        }


        teamInput.addEventListener('input', updateCalculator);

        hoursInput.addEventListener('input', updateCalculator);

        effInput.addEventListener('input', updateCalculator);

    </script>
